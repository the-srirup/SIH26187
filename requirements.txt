# IBVAP — Intelligent Border Video Analytics Platform
# ---------------------------------------------------------------------------
# Core runtime. Install with:
#     pip install -r requirements.txt
#
# GPU NOTE ------------------------------------------------------------------
# The default PyPI `torch` is a CPU-only build. On a machine with an NVIDIA
# GPU install the CUDA build FIRST, or inference silently runs on the CPU at
# roughly a third of the speed:
#
#     pip install --index-url https://download.pytorch.org/whl/cu126 \
#                 torch torchvision
#
# torchvision must come from the same CUDA index as torch — a CPU torchvision
# paired with a CUDA torch fails at NMS with "torchvision::nms not implemented
# for CUDA". IBVAP detects that at start-up and falls back to CPU rather than
# crashing, and /api/system/info reports which device is really in use.
# ---------------------------------------------------------------------------

# Both degrade gracefully: if the import fails the feature reports itself as
# unavailable in /api/system/info and the surveillance pipeline is unaffected.
insightface>=0.7.3             # SCRFD face detection + ArcFace embeddings
onnxruntime>=1.19.0            # InsightFace runtime — see the GPU note below
easyocr>=1.7.0                 # ANPR optical character recognition
yt-dlp>=2024.1.1               # resolves YouTube / YouTube Live to a stream URL

# FACE-STAGE GPU NOTE --------------------------------------------------------
# `onnxruntime` above is the CPU build, and InsightFace will silently use it
# even on a CUDA machine — /api/system/info reports the providers actually in
# use, and the face stage costs ~19 ms/frame on CPU versus a few ms on GPU.
# On an NVIDIA host, replace it:
#
#     pip uninstall -y onnxruntime && pip install onnxruntime-gpu
#
# The platform runs correctly either way; this is throughput, not correctness.
